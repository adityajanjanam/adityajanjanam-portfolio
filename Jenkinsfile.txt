pipeline {
    agent any

    stages {
        stage("Checkout") {
            steps {
                checkout scm
            }
        }

        stage("Install Dependencies") {
            steps {
                bat 'npm ci'
            }
        }

        stage("Build React App") {
            steps {
                bat 'npm run build'
            }
        }

        stage("Deploy to Firebase") {
            steps {
                withCredentials([
                    string(credentialsId: 'FIREBASE_CI_TOKEN', variable: 'FIREBASE_TOKEN'),
                    string(credentialsId: 'FIREBASE_PROJECT_ID', variable: 'FIREBASE_PROJECT')
                ]) {
                    bat """
                        npm install -g firebase-tools
                        firebase deploy --only hosting --project %FIREBASE_PROJECT% --token %FIREBASE_TOKEN%
                    """
                }
            }
        }
    }
}
